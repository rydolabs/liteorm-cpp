cmake_minimum_required(VERSION 3.10)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(tinyorm_cpp11 0.7 CONFIG REQUIRED)
add_executable(tinyorm_legacy_consumer main.cpp)
target_link_libraries(tinyorm_legacy_consumer PRIVATE tinyorm::tinyorm)
