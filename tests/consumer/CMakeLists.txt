cmake_minimum_required(VERSION 3.10)
project(liteorm_installed_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(liteorm_cpp 0.7 CONFIG REQUIRED)
add_executable(liteorm_installed_consumer main.cpp)
target_link_libraries(liteorm_installed_consumer PRIVATE liteorm::liteorm)
